struct Point {
    x: f64,
    y: f64,
}

impl Point {
    fn norm(&self) -> f64 {
        (self.x.powi(2) + self.y.powi(2)).sqrt()
    }
}

fn main() {
    let p = Point { x: 3.0, y: 4.0 };
    let names = vec!["Maria", "José"];

    for name in &names {
        println!("Hello, {name}!");
    }
    println!("norm = {}", p.norm());
}
      

from dataclasses import dataclass
import math

@dataclass
class Point:
    x: float
    y: float

    def norm(self) -> float:
        return math.hypot(self.x, self.y)
